<script>

    amount: 99.00, // Exact amount in rupees
    currency: "INR",

    description: "Event Registration Fee",
    customer: {
      name: "Mohit Verma",
      email: "mohit@example.com",
      phone: "9876543210"
    },
    handler: function(response) {
      // Called automatically once payment is verified!
      console.log("Verified Payment Response:", response);
      alert("Payment Verified Successfully! UTR: " + response.utr);
      // Fulfill registration or redirect
      window.location.href = "/registration-success?utr=" + response.utr;
    },
    onClose: function() {
      console.log("Customer closed the checkout modal");
    }
  });
}
</script>
